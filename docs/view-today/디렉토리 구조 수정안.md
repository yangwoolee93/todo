```
src/renderer/src/

pages/today/
  TodayPage.tsx          ← App에서 view==="today"일 때 마운트
  index.ts

widgets/today/
  TodayWidget.tsx        ← 레이아웃 조립 (헤더 + 목록 영역)
  index.ts

features/today/
  ui/
    TodayHeader.tsx      ← 상단 card (날짜·전/다음·다른날짜·건수·추가)
    ChevronIcons.tsx     ← (선택) 아이콘
  hooks/
    useTodayHeader.ts    ← (선택) isToday, shift, goToday
  index.ts               ← export { TodayHeader }

features/todo/
  ui/
    TodoList.tsx         ← components/TodoList.tsx 이동
    TodoItemMenu.tsx
    TodoStatusIcon.tsx
    EditTodoModal.tsx
    DeleteConfirmModal.tsx
    DeleteBatchModal.tsx
  hooks/
    useTodos.ts          ← hooks/useTodos.ts 이동
  model/
    useTodoStore.ts      ← (나중) useTodos 대체
  index.ts               ← export { TodoList, useTodos, ... }

features/todo-add/         ← (선택) 추가 모달만 분리할 때
  ui/
    AddTodoModal.tsx
  index.ts

shared/
  ui/                    ← 범용 Button, Modal shell 등
  lib/
    dateUtils.ts
  types/
    views.ts

stores/                    ← (당분간) 전역 UI
  useUIStore.ts            ← view, activeDate, (나중) 모달 open

app/
  App.tsx                  ← AppShell + view 분기 + pages 호출
  AppShell.tsx             ← layout/AppShell 이동
```

```
App
 └─ pages/today/TodayPage          (todos·handlers 수신)
      └─ widgets/today/TodayWidget
           ├─ features/today/TodayHeader     → useUIStore (activeDate)
           └─ features/todo/TodoList          → props 또는 useTodoStore
```

```
components/views/TodayView.tsx  →  pages/today/TodayPage.tsx
                                 +  widgets/today/TodayWidget.tsx
                                 +  features/today/ui/TodayHeader.tsx

components/TodoList.tsx         →  features/todo/ui/TodoList.tsx
hooks/useTodos.ts               →  features/todo/hooks/useTodos.ts
stores/useUIStore.ts            →  shared 또는 app 레벨 유지
```
